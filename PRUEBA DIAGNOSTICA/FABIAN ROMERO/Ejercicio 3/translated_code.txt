#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

// Structure to associate C keywords with their translation
typedef estructura {
    const caracter *english;
    const caracter *spanish;
    entero counter;
} Keyword;

// Table containing the main C reserved keywords
static Keyword table[] = {
    {"entero", "entero", 0},
    {"caracter", "caracter", 0},
    {"flotante", "flotante", 0},
    {"doble", "doble", 0},
    {"si", "si", 0},
    {"sino", "sino", 0},
    {"para", "para", 0},
    {"mientras", "mientras", 0},
    {"retornar", "retornar", 0},
    {"vacio", "vacio", 0},
    {"estructura", "estructura", 0},
    {"romper", "romper", 0},
    {"segun", "segun", 0},
    {"caso", "caso", 0},
    {"por_defecto", "por_defecto", 0}
};
#define NUM_KEYWORDS (sizeof(table) / sizeof(table[0]))

// 1. Dynamic file loading into memory using malloc and realloc
caracter* load_file(const caracter *path, size_t *size_out) {
    FILE *f = fopen(path, "r");
    si (!f) retornar NULL;

    size_t capacity = 512, size = 0;
    caracter *buffer = malloc(capacity);
    si (!buffer) { fclose(f); retornar NULL; }

    entero c;
    mientras ((c = fgetc(f)) != EOF) {
        si (size + 1 >= capacity) {
            capacity *= 2;
            caracter *temp = realloc(buffer, capacity);
            si (!temp) { free(buffer); fclose(f); retornar NULL; }
            buffer = temp;
        }
        buffer[size++] = (caracter)c;
    }
    buffer[size] = '\0';
    fclose(f);
    *size_out = size;
    retornar buffer;
}

entero main(entero argc, caracter **argv) {
    si (argc < 2) {
        printf("Usage: %s source_file.c\n", argv[0]);
        retornar 1;
    }

    size_t size;
    caracter *content = load_file(argv[1], &size);
    si (!content) {
        printf("Error: Could not read the file.\n");
        retornar 1;
    }

    FILE *f_trad = fopen("translated_code.txt", "w");
    FILE *f_rep = fopen("keywords_report.txt", "w");

    fprintf(f_rep, "=== KEYWORDS REPORT ===\n");
    fprintf(f_rep, "Analyzed file: %s (%zu bytes loaded into dynamic memory)\n\n", argv[1], size);

    // 2. Parse the dynamic buffer character by character
    size_t i = 0;
    mientras (i < size) {
        si (isalpha(content[i]) || content[i] == '_') {
            size_t start = i;
            mientras (i < size && (isalnum(content[i]) || content[i] == '_')) {
                i++;
            }
            size_t len = i - start;
            caracter *word = malloc(len + 1);
            memcpy(word, &content[start], len);
            word[len] = '\0';

            // Check si the word is a reserved keyword
            entero found = 0;
            para (size_t j = 0; j < NUM_KEYWORDS; j++) {
                si (strcmp(word, table[j].english) == 0) {
                    table[j].counter++;
                    fprintf(f_trad, "%s", table[j].spanish);
                    fprintf(f_rep, "-> '%s' translated to '%s'\n", table[j].english, table[j].spanish);
                    found = 1;
                    romper;
                }
            }
            si (!found) {
                fprintf(f_trad, "%s", word);
            }
            free(word);
        } sino {
            fputc(content[i], f_trad);
            i++;
        }
    }

    // 3. Write summary to the report
    fprintf(f_rep, "\n=== OCCURRENCE SUMMARY ===\n");
    entero total = 0;
    para (size_t j = 0; j < NUM_KEYWORDS; j++) {
        si (table[j].counter > 0) {
            fprintf(f_rep, "%s -> %s : %d times\n", table[j].english, table[j].spanish, table[j].counter);
            total += table[j].counter;
        }
    }
    fprintf(f_rep, "\nTotal reserved keywords found: %d\n", total);

    // Close files and free the main dynamic memory buffer
    fclose(f_trad);
    fclose(f_rep);
    free(content);

    printf("Process successful!\nFiles generated: 'translated_code.txt' and 'keywords_report.txt'\n");
    retornar 0;
}
